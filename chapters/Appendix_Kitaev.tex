\section{Transforming the Kitaev chain Hamiltonian to Majorana operators} 
\label{Appendix:Hamiltonian_Transformation_Majorana_ops}
This appendix shows how transform the original Kitaev chain Hamiltonian expressed in fermionic operators $a_j$ and $a_j^\dagger$ to being expressed in terms of the majorana operators $c_{2j-1}$ and $c_2j$. 

Starting with the majorana operators
\begin{align}
    c_{2j-1} &= e^{i \frac{\theta}{2}} a_j + e^{-i \frac{\theta}{2}}a_j^\dagger \\
    c_{2j} &= -ie^{i \frac{\theta}{2}} a_j + ie^{-i \frac{\theta}{2}}a_j^\dagger.
\end{align}
We can invert them, leading to four expressions
\begin{align}
    a_j &= e^{- i \frac{\theta}{2}} c_{2j-1} - e^{-i \theta} a_j^\dagger \\
    &= i e^{- i \frac{\theta}{2}} c_{2j} + e^{- i \theta} a_j^\dagger \\
    a_j^\dagger &= e^{i \frac{\theta}{2}} c_{2j-1} - e^{i\theta} a_j \\
    &= - i e^{i\frac{\theta}{2}} c_{2j} + e^{i \theta} a_j.
\end{align}
Combining the expressions for $a_j$, and for $a_j^\dagger$ we find
\begin{align}
    a_j &= \frac{1}{2} \left( e^{- i \frac{\theta}{2}} c_{2j-1} - e^{- i \theta} a_j^\dagger + e^{- i \theta} a_j^\dagger + i e^{- i \frac{\theta}{2}} c_2j \right) \\
    &= \frac{1}{2} e^{-i \frac{\theta}{2}} \left( c_{2j-1} + i c_{2j} \right)\\
    a_j^\dagger &= \frac{1}{2} \left( e^{i \frac{\theta}{2}} c_{2j-1} - e^{i\theta} a_j + e^{i \theta} a_j - i e^{i\frac{\theta}{2}} c_{2j}   \right) \\
    &= \frac{1}{2} e^{i \frac{\theta}{2}} \left( c_{2j-1} - i c_{2j}  \right).
\end{align}
We now substitute these expressions into the original Hamiltonian
\begin{align}
    H =& \sum_j\left[ - \omega \left(a_j^\dagger a_{j+1} + a_{j+1}^\dagger a_j  \right) - \mu \left(a_j^\dagger  a_j - \frac{1}{2} \right) + \Delta a_j a_{j+1} + \Delta^* a_{j+1}^\dagger a_j^\dagger \right]\\
    &= -\omega \sum_j  \frac{1}{2} e^{i \frac{\theta}{2}} \left( c_{2j-1} - i c_{2j}  \right) \frac{1}{2} e^{-i \frac{\theta}{2}} \left( c_{2(j+1)-1} + i c_{2(j+1)} \right) \\
    &-\omega \sum_j \frac{1}{2} e^{i \frac{\theta}{2}} \left( c_{2(j+1)-1} - i c_{2(j+1)}  \right) \frac{1}{2} e^{-i \frac{\theta}{2}} \left( c_{2j-1} + i c_{2j} \right)\\
    &- \mu \sum_j \left(\frac{1}{2} e^{i \frac{\theta}{2}} \left( c_{2j-1} - i c_{2j}  \right) \frac{1}{2} e^{-i \frac{\theta}{2}} \left( c_{2j-1} + i c_{2j} \right)  - \frac{1}{2}\right) \\
    &+ \Delta \sum_j \frac{1}{2} e^{-i \frac{\theta}{2}} \left( c_{2j-1} + i c_{2j} \right) \frac{1}{2} e^{-i \frac{\theta}{2}} \left( c_{2(j+1)-1} + i c_{2(j+1)} \right) \\
    &+ \Delta^* \sum_j \frac{1}{2} e^{i \frac{\theta}{2}} \left( c_{2(j+1)-1} - i c_{2(j+1)}  \right) \frac{1}{2} e^{i \frac{\theta}{2}} \left( c_{2j-1} - i c_{2j}  \right)\\
    &= - \frac{\omega}{4} \sum_j \left( c_{2j-1} c_{2j + 1} + i c_{2j-1} c_{2j + 2} - i c_{2j} c_{2j + 1} + c_{2j} c_{2j + 2} \right) \\
    &- \frac{\omega}{4} \sum_j \left( c_{2j+1} c_{2j - 1} + i c_{2j + 1} c_{2j} - i c_{2j + 2} c_{2j - 1} + c_{2j + 2} c_{2j}\right)\\
    & -\frac{\mu}{4} \sum_j \left( c_{2j-1}^2 + ic_{2j-1} c_{2j} - i c_{2j} c_{2j-1} + c_{2j}^2 - 2 \right) \\
    &+ \frac{|\Delta|}{4}e^{i\theta} \sum_j e^{-i \theta} \left( c_{2j-1} c_{2j+1} + i c_{2j-1} c_{2j+2} + i c_{2j} c_{2j+1} - c_{2j} c_{2j+2} \right) \\
    &+ \frac{|\Delta|}{4} e^{-i\theta} \sum_j e^{i\theta}\left( c_{2j+1} c_{2j-1} - i c_{2j + 1} c_{2j} - i c_{2j+2} c_{2j-1} - c_{2j+2} c_{2j}\right) \\
    &= - \frac{\omega}{4} \sum_j \left( 0 + 2 i c_{2j-1} c_{2j+2} - 2i c_{2j}c_{2j+1} + 0\right) \\
    &- \frac{\mu}{4} \sum_j \left( 2i c_{2j-1}c_{2j} + 2 - 2\right)\\
    &+ \frac{|\Delta|}{4}\sum_j \left( 0 + 2i c_{2j-1}c_{2j+2} + 2i c_{2j} c_{2j+1} + 0  \right) \\
    &= \frac{i}{2} \sum_j \left[-\mu c_{2j-1} c_{2j} + \left(\omega + \left|\Delta \right|  \right) c_{2j} c_{2j+1} + \left(-\omega + \left| \Delta \right| \right) c_{2j-1} c_{2j+2} \right]
\end{align}
Where we used $\Delta = |\Delta|e^{i\theta}$ and the anticommutation relations
\begin{align}
    \left\{ c_l, c_m \right\} &= c_l c_m + c_m c_l = 2\delta_{lm} \\
    c_l c_m &= - c_m c_l (m \neq l)\\
    c_l^2 &= 1.
\end{align}

\section{Transforming the Kitaev chain Hamiltonian to momentum space}    
\label{Appendix:Hamiltonian_Transformation_momentum_space}
Taking inspiration from \cite{aliceaNewDirectionsPursuit2012} we start from the Kitaev chain Hamiltonian \ref{eq:Original_Kitaev_Hamiltonian}. Imposing periodic boundary conditions we can Fourier transform the Hamiltonian to momentum space $q$. We recall the Fourier transform
\begin{align}
    a_j = \frac{1}{\sqrt{N}} \sum_q e^{iqj}a_q, \hspace{4mm} a_j^\dagger = \frac{1}{\sqrt{N}} \sum_q e^{-iqj}a_q^\dagger
\end{align}
where $q$ runs over the Brillouin zone $[-\pi, \pi]$. Using this we can transform each term in the Hamiltonian individually for simplicity. Starting with the on-site term
\begin{align}
    H_\mu &= -\mu \sum_j \left(a_j^\dagger a_j - \frac{1}{2} \right) \\
    &=  -\mu \sum_j \left( \left( \frac{1}{\sqrt{N}} \sum_q e^{-iqj}a_q^\dagger \right) \left( \frac{1}{\sqrt{N}} \sum_{q'} e^{iq'j}a_{q'} \right) - \frac{1}{2}\right) \\
    &= -\mu \frac{1}{N} \sum_{j q q'} e^{i(q'-q)j} a_q^\dagger a_{q'} + \frac{\mu N}{2}\\
    &= -\frac{\mu}{N} \sum_{qq'} N \delta_{qq'} a_q^\dagger a_{q'} + \frac{\mu N}{2}\\
    &= -\mu \sum_q  a_q^\dagger a_q + \frac{\mu N}{2}.
\end{align}
Where we used a property of the periodic boundary conditions $\sum_j e^{i(q'-q)j} = N \delta_{qq'}$. Continuing with the hopping term
\begin{align}
    H_\omega &= -\omega \sum_j \left(a_j^\dagger a_{j+1} + a_{j+1}^\dagger a_j \right)
\end{align}
where we again split the calculation of the terms; the first term becomes
\begin{align}
    H_\omega^1 &= -\omega \sum_j \left( \frac{1}{\sqrt{N}} \sum_q e^{-iqj}a_q^\dagger \right) \left( \frac{1}{\sqrt{N}} \sum_{q'} e^{iq'(j+1)}a_{q'} \right)\\
    &= - \frac{\omega}{N} \sum_{j q q'} e^{i(q'-q)j}e^{iq'} a_q^\dagger a_{q'}\\
    &= -\frac{\omega}{N} \sum_{qq'} N \delta_{qq'}e^{iq'} a_q^\dagger a_{q'}\\
    &= -\omega \sum_q e^{iq} a_q^\dagger a_{q}.
\end{align}
And the second term similarly becomes
\begin{align}
    H_\omega^2 &= -\omega \sum_q e^{-iq} a_q^\dagger a_q.
\end{align}
Combining the hopping terms we get
\begin{align}
    H_\omega &= H_\omega^1 + H_\omega^2 \\
    &= -\omega \sum_q \left( e^{iq} + e^{-iq} \right) a_q^\dagger a_q\\
    &= - 2 \omega \sum_q \cos(q) a_q^\dagger a_q.
\end{align}
We now consider the pairing term
\begin{align}
    H_\Delta &= \sum_j \left( \Delta a_j a_{j+1} + \Delta^* a_{j+1}^\dagger a_j^\dagger \right)\\
\end{align}
where we again split the calculation for simplicity
\begin{align}
    H_\Delta^1 &= \Delta \sum_j \left( \frac{1}{\sqrt{N}} \sum_q e^{iqj}a_q \right) \left( \frac{1}{\sqrt{N}} \sum_{q'} e^{iq'(j+1)}a_{q'} \right)\\
    &= \frac{\Delta}{N} \sum_{jqq'} e^{i (q + q') j} e^{iq'} a_q a_{q'}\\
    &= \frac{\Delta}{N} \sum_{qq'} N \delta_{q (-q')} e^{iq'} a_q a_{q'}\\
    &= \Delta \sum_q e^{-iq} a_q a_{-q}.
\end{align}
And the second term similarly becomes
\begin{align}
    H_\Delta^2 &= \Delta^* \left(  \frac{1}{\sqrt{N}} \sum_q e^{-iq(j+1)}a_q^\dagger  \right) \left(  \frac{1}{\sqrt{N}} \sum_{q'} e^{-iq'j}a_{q'}^\dagger    \right)\\
    &= \frac{\Delta^*}{N} \sum_{jqq'} e^{-i(q+q')j} e^{-iq} a_q^\dagger a_{q'}^\dagger \\
    &= \frac{\Delta^*}{N} \sum_{qq'} N \delta_{q(-q')} e^{-iq} a_q^\dagger a_{q'}^\dagger \\
    &= \Delta^* \sum_{q} e^{-iq} a_q^\dagger a_{-q}^\dagger.
\end{align}
Combining all terms we get the momentum-space Hamiltonian
\begin{align}
    H &= \sum_q \left[ -\mu \left( a_q^\dagger a_q \right) - 2\omega \cos(q)a_q^\dagger a_q + \Delta e^{-iq} a_q a_{-q} + \Delta^* e^{-iq} a_q^\dagger a_{-q}^\dagger \right] + \frac{\mu N}{2}\\
      &= \sum_q \left[ -\left( \mu + 2 \omega \cos(q)\right) a_q^\dagger a_q + \Delta e^{-iq} a_q a_{-q} + \Delta^* e^{-iq} a_q^\dagger a_{-q}^\dagger \right]
\end{align}
Where the last term $+ \frac{\mu N}{2}$ can be ignored as it does not affect the spectrum, eigenstates, or any observable that depends on the energy differences.
